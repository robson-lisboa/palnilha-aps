'----------------------------------------------------------
' Excecoes em memoria (uma por data; a ultima definida prevalece)
'----------------------------------------------------------
Private Function IdxExc(ByVal d As Double) As Long
    Dim i As Long
    For i = 1 To mNExc
        If mExcData(i) = d Then IdxExc = i: Exit Function
    Next i
End Function

Private Sub DefinirExc(ByVal d As Double, ByVal st As String, ByVal hor As String)
    Dim i As Long
    i = IdxExc(d)
    If i = 0 Then
        mNExc = mNExc + 1
        If mNExc > UBound(mExcData) Then
            ReDim Preserve mExcData(1 To mNExc + 8)
            ReDim Preserve mExcSt(1 To mNExc + 8)
            ReDim Preserve mExcHor(1 To mNExc + 8)
        End If
        i = mNExc
    End If
    mExcData(i) = d: mExcSt(i) = st: mExcHor(i) = hor
End Sub

Private Sub RemoverExc(ByVal d As Double)
    Dim i As Long
    i = IdxExc(d)
    If i = 0 Then Exit Sub
    mExcData(i) = mExcData(mNExc): mExcSt(i) = mExcSt(mNExc): mExcHor(i) = mExcHor(mNExc)
    mNExc = mNExc - 1
End Sub

' 0 = padrao com producao | 1 = padrao sem producao | 2 = LIBERADA por excecao | 3 = BLOQUEADA por excecao
Private Function EstadoData(ByVal d As Double) As Long
    Dim i As Long
    i = IdxExc(d)
    If i > 0 Then
        If mExcSt(i) = "S" Then EstadoData = 2 Else EstadoData = 3
    ElseIf chkDia(Weekday(d, vbMonday)).Value Then
        EstadoData = 0
    Else
        EstadoData = 1
    End If
End Function

'----------------------------------------------------------
' Calendario mensal
'----------------------------------------------------------
Private Sub AtualizarGrade()
    Dim off As Long, nd As Long, i As Long, dia As Long, d As Double, est As Long

    If Not mMontado Then Exit Sub

    lblMes.Caption = NomeMes(mMes) & " / " & mAno
    off = Weekday(DateSerial(mAno, mMes, 1), vbMonday) - 1
    nd = Day(DateSerial(mAno, mMes + 1, 0))

    For i = 1 To 42
        dia = i - off
        With cel(i)
            If dia >= 1 And dia <= nd Then
                d = CDbl(DateSerial(mAno, mMes, dia))
                est = EstadoData(d)
                .Enabled = True
                .Caption = IIf(d = mSel, ">" & dia & "<", CStr(dia))
                .Font.Bold = (est >= 2) Or (d = mSel)
                Select Case est
                    Case 0: .BackColor = RGB(232, 245, 233): .ForeColor = RGB(30, 30, 30)
                    Case 1: .BackColor = RGB(238, 238, 238): .ForeColor = RGB(120, 120, 120)
                    Case 2: .BackColor = RGB(76, 175, 80): .ForeColor = RGB(255, 255, 255)
                    Case Else: .BackColor = RGB(229, 57, 53): .ForeColor = RGB(255, 255, 255)
                End Select
            Else
                .Caption = ""
                .Enabled = False
                .BackColor = RGB(250, 250, 250)
            End If
        End With
    Next i
    AtualizarSelecao
End Sub

Private Sub AtualizarSelecao()
    Dim wd As Long, est As Long, i As Long, t As String, hor As String

    cmdPadrao.Enabled = (mSel > 0)
    cmdLiberar.Enabled = (mSel > 0)
    cmdBloquear.Enabled = (mSel > 0)
    If mSel <= 0 Then
        lblSel.Caption = "Nenhuma data selecionada. Clique em uma data do calend" & ChrW(225) & "rio."
        Exit Sub
    End If

    wd = Weekday(mSel, vbMonday)
    est = EstadoData(mSel)
    t = "Data: " & NomeDia(wd) & ", " & Format$(mSel, "dd/mm/yyyy") & vbCr
    If chkDia(wd).Value Then
        t = t & "Semana padr" & ChrW(227) & "o: COM produ" & ChrW(231) & ChrW(227) & "o. "
    Else
        t = t & "Semana padr" & ChrW(227) & "o: SEM produ" & ChrW(231) & ChrW(227) & "o. "
    End If
    Select Case est
        Case 0, 1: t = t & "Esta data segue o padr" & ChrW(227) & "o."
        Case 2
            i = IdxExc(mSel)
            hor = mExcHor(i)
            If Len(hor) = 0 Then hor = Trim$(txtI(wd).Text) & ";" & Trim$(txtF(wd).Text)
            t = t & "LIBERADA por exce" & ChrW(231) & ChrW(227) & "o (" & Replace(hor, ";", " - ") & ")."
        Case Else: t = t & "BLOQUEADA por exce" & ChrW(231) & ChrW(227) & "o."
    End Select
    lblSel.Caption = t
End Sub

Private Sub DiaClick(ByVal i As Long)
    Dim off As Long, dia As Long
    off = Weekday(DateSerial(mAno, mMes, 1), vbMonday) - 1
    dia = i - off
    If dia < 1 Or dia > Day(DateSerial(mAno, mMes + 1, 0)) Then Exit Sub
    mSel = CDbl(DateSerial(mAno, mMes, dia))
    AtualizarGrade
End Sub

Private Sub cmdPrev_Click()

Private Sub cmdNext_Click()

    If mMes > 12 Then mMes = 1: mAno = mAno + 1
    mSel = 0

End Sub

Private Sub cmdPadrao_Click()

Private Sub cmdLiberar_Click()

'----------------------------------------------------------
Private Sub cmdSalvar_Click()
    Dim i As Long, j As Long, s As String, linhas() As String, p() As String, saida As String
    Dim h1 As Double, h2 As Double, oee As Double, t As String, ordem() As Long, tmp As Long

    ' semana: horario de cada dia validado (dia sem producao com horario invalido volta ao padrao 06:00-22:00)
    For i = 1 To 7
        h1 = APS_ParaHora(txtI(i).Text)
        h2 = APS_ParaHora(txtF(i).Text)
        If chkDia(i).Value Then
            If h1 < 0 Or h2 < 0 Then
                MsgBox "Hor" & ChrW(225) & "rio inv" & ChrW(225) & "lido em " & NomeDia(i) & " (use hh:mm).", vbExclamation, APS_TITULO
                Exit Sub
            End If
            If h2 > 0 And h2 <= h1 Then
                MsgBox "O fim deve ser maior que o in" & ChrW(237) & "cio em " & NomeDia(i) & ".", vbExclamation, APS_TITULO
                Exit Sub
            End If
        Else
            If h1 < 0 Or h2 < 0 Or (h2 > 0 And h2 <= h1) Then h1 = 6# / 24#: h2 = 22# / 24#
        End If
        s = s & i & ";" & IIf(chkDia(i).Value, "S", "N") & ";" & Format$(h1, "hh:nn") & ";" & Format$(h2, "hh:nn") & "|"
    Next i
    s = Left$(s, Len(s) - 1)

    ' excecoes por data (ordenadas por data)
    saida = ""
    If mNExc > 0 Then
        ReDim ordem(1 To mNExc)
        For i = 1 To mNExc
            ordem(i) = i
        Next i
        For i = 1 To mNExc - 1
            For j = i + 1 To mNExc
                If mExcData(ordem(j)) < mExcData(ordem(i)) Then tmp = ordem(i): ordem(i) = ordem(j): ordem(j) = tmp
            Next j
        Next i
        For i = 1 To mNExc
            j = ordem(i)
            If Len(saida) > 0 Then saida = saida & "|"
            saida = saida & Format$(mExcData(j), "dd/mm/yyyy") & ";" & mExcSt(j)
            If mExcSt(j) = "S" And Len(mExcHor(j)) > 0 Then saida = saida & ";" & mExcHor(j)
        Next i
    End If

    ' paradas
    t = ""
    linhas = Split(Replace(txtPar.Text, vbLf, ""), vbCr)
    For i = 0 To UBound(linhas)
        If Len(Trim$(linhas(i))) > 0 Then
            p = Split(linhas(i), ";")
            If UBound(p) < 1 Then
                MsgBox "Parada inv" & ChrW(225) & "lida (linha " & (i + 1) & "): " & linhas(i), vbExclamation, APS_TITULO
                Exit Sub
            End If
            h1 = APS_ParaDataHora(p(0))
            h2 = APS_ParaDataHora(p(1))
            If h1 < 0 Or h2 <= h1 Then
                MsgBox "Parada inv" & ChrW(225) & "lida (linha " & (i + 1) & "): " & linhas(i), vbExclamation, APS_TITULO
                Exit Sub
            End If
            If Len(t) > 0 Then t = t & "|"
            t = t & Trim$(linhas(i))
        End If
    Next i

    ' OEE padrao
    oee = Val(Replace(Replace(Trim$(txtOEEPad.Text), "%", ""), ",", "."))
    If oee <= 0 Or oee > 100 Then
        MsgBox "OEE padr" & ChrW(227) & "o deve estar entre 1 e 100.", vbExclamation, APS_TITULO
        Exit Sub
    End If

    APS_CfgGravar "CAL_SEM", s
    APS_CfgGravar "CAL_EXC", saida
    APS_CfgGravar "CAL_PAR", t
    APS_CfgGravar "OEE_PADRAO", Replace(CStr(oee), ",", ".")
    APS_CalRecarregar
    Unload Me
    APS_Atualizar True
End Sub
